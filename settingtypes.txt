# Walk speed multiplier while sprinting.
mese_combat.sprint_speed (Sprint speed multiplier) float 1.3 1.0 2.0

# Field of view multiplier while sprinting. 1 disables the effect.
mese_combat.sprint_fov (Sprint FOV multiplier) float 1.08 1.0 1.3

# Damage multiplier for a critical hit (full charge while falling).
mese_combat.crit_multiplier (Critical hit multiplier) float 1.5 1.0 3.0

# Charge fraction that counts as a full-charge hit.
mese_combat.full_charge (Full charge threshold) float 0.9 0.5 1.0

# Horizontal knockback for a full-charge hit (node/s).
mese_combat.kb_base (Base knockback) float 5.5 0.0 20.0

# Extra horizontal knockback for a sprint hit (node/s).
mese_combat.kb_sprint (Sprint-hit extra knockback) float 3.5 0.0 20.0

# Upward knockback when the target is on the ground (node/s).
mese_combat.kb_up (Vertical knockback) float 4.0 0.0 20.0

# Show the charge bar, damage numbers, and combo counter.
mese_combat.hud (Combat indicators) bool true

# Hits on mobs with a mese_combat weapon use charge, crits and sprint hits too.
mese_combat.mobs (Combat system on mobs) bool true

# Damage multiplier for weapon hits on mobs. Use it when weapons are tuned for
# more player HP than your mobs have. 0.4 turns a 14 damage hit into 6.
mese_combat.mob_damage (Damage multiplier on mobs) float 1.0 0.05 5.0
